04.Day Name Using Switch

let day = 3

switch day {
case 1: print("Monday")
case 2: print("Tuesday")
case 3: print("Wednesday")
case 4: print("Thursday")
case 5: print("Friday")
case 6: print("Saturday")
case 7: print("Sunday")
default: print("Invalid")
}

09.Largest of Three Numbers Using Nested If

let a = 10
let b = 25
let c = 15

if a > b {
    if a > c {
        print(a)
    } else {
        print(c)
    }
} else {
    if b > c {
        print(b)
    } else {
        print(c)
    }
}

11.Sum of Digits Using While Loop

var number = 1234
var sum = 0

while number > 0 {
    sum += number % 10
    number /= 10
}

print("Sum = \(sum)")

12.Fibonacci Series up to n Terms

let n = 10
var a = 0
var b = 1

for _ in 1...n {
    print(a)
    let temp = a + b
    a = b
    b = temp
}

13.Check Prime Number and Print Primes from 1 to 50

let number = 7
var isPrime = number > 1

if number > 2 {
    for i in 2..<number {
        if number % i == 0 {
            isPrime = false
            break
        }
    }
}

print(isPrime ? "Prime" : "Not Prime")

for n in 2...50 {
    var prime = true

    if n > 2 {
        for i in 2..<n {
            if n % i == 0 {
                prime = false
                break
            }
        }
    }

    if prime {
        print(n)
    }
}

14.Simple Calculator Using Switch

let a = 10.0
let b = 5.0
let op = "+"

switch op {
case "+":
    print(a + b)
case "-":
    print(a - b)
case "*":
    print(a * b)
case "/":
    if b != 0 {
        print(a / b)
    } else {
        print("Division by zero")
    }
default:
    print("Invalid Operator")
}

15.FizzBuzz from 1 to 30

for i in 1...30 {
    if i % 3 == 0 && i % 5 == 0 {

    } else if i % 3 == 0 {
        print("Fizz")
    } else if i % 5 == 0 {
        print("Buzz")

17.Palindrome and Armstrong Number

let number = 153

// Palindrome check
var temp = number
var reverse = 0

while temp > 0 {
    reverse = reverse * 10 + temp % 10
    temp /= 10
}

print(reverse == number ? "Palindrome" : "Not Palindrome")

// Armstrong check
let digits = String(number).count
temp = number
var sum = 0

while temp > 0 {
    let digit = temp % 10
    sum += Int(pow(Double(digit), Double(digits)))
    temp /= 10
}

print(sum == number ? "Armstrong" : "Not Armstrong")

Skip even rows using continue

let n = 5

for i in 1...n {
    if i % 2 == 0 {
        continue
    }

    for _ in 1...i {
        print("*", terminator: " ")
    }
    print()
}

18.BMI Categories Using Switch and Where

let bmi = 24.5

switch bmi {
case ..<18.5:
    print("Underweight")
case 18.5..<25:
    print("Normal")
case 25..<30:
    print("Overweight")
case 30... where bmi >= 30:
    print("Obese")
default:
    print("Invalid BMI")
}
